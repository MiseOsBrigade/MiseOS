  mode: advisory         # 'advisory' | 'enforced'
  environment: github    # later: gitlab, bitbucket, local, etc.
  repoprofile: service  # 'service' | 'library' | 'infra' | 'data' | 'docs'

  policies:
    pr:
      enabled: true
      maxchangedfiles: 50
      maxlineschanged: 800
      requiretestswhenpathsmatch:
        - "src/"
        - "services/"
      riskypaths:
        - "infra/"
        - "auth/"
        - ".github/workflows/"

    stewardship:
      enabled: true
      stalebranchdays: 30
      autocleanup: false    # only propose cleanup PRs, don’t auto‑delete
      requirereviewers: 1

    security:
      enabled: true
      requiresignedcommits: false
      blockforcepushtodefault: true

  modules:
    pr: true
    brigade: true
    saltshakers: true
    locker: true
    gpu: false
`

Why this is safe:

• It doesn’t touch existing workflows, scripts or infra.
• Modules read from it and only post checks / comments while in mode: advisory.
• If someone removes mise.yaml, the repo just loses Mise features; no bricking.

1.2. “Non‑jeopardizing” installation pattern

In any existing repo:

Add mise.yaml.
Add one minimal workflow (GitHub Actions example):

`yaml
.github/workflows/mise-orchestra.yml
name: Mise DevOps Orchestra

on:
  pullrequest:
  push:
    branches: [ main, master ]
  workflowdispatch:

jobs:
  mise-orchestra:
    runs-on: ubuntu-latest

    permissions:
      contents: read
      pull-requests: write
      checks: write

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Run Mise Orchestra (advisory mode)
        uses: mise-org/mise-orchestra-action@v1
        with:
          configpath: mise.yaml
          mode: advisory
`

Design that action to:

• read mise.yaml
• call out (via CLI or HTTP) to your Mise‑Maestro controller
• receive back a list of checks to run + comments to post
• run them, but:
  - never modify files, unless explicitly configured
  - never fail the build while in advisory mode; only set neutral/warning statuses

Once teams trust it, they can flip:

`yaml

`ts
import { miseOrchestraRoster } from "./mise-orchestra-roster";
import { miseExecutiveTier } from "./mise-orchestra-executives";
import { DepartmentId, MiseCharacter, Rarity } from "./mise-orchestra-types";

export const departmentOrder: DepartmentId[] = [
  "pipeline",
  "platform",
  "reliability",
  "security",
  "data-insight"
];

export const departmentLabels: Record<DepartmentId, string> = {
  pipeline: "Pipeline Kitchen",
  platform: "Platform Kitchen",
  reliability: "Reliability Kitchen",
  security: "Security Kitchen",
  "data-insight": "Data & Insight Kitchen"
};

export const departmentTaglines: Record<DepartmentId, string> = {
  pipeline: "Build. Test. Deploy. Repeat.",
  platform: "The stage where services live.",
  reliability: "Observe. Measure. Improve.",
  security: "Protect the kitchen.",
  "data-insight": "Learn. Optimize. Evolve."
};

export function getAllCharacters(): MiseCharacter[] {
  return [...miseOrchestraRoster];
}

export function getCharacterById(id: string): MiseCharacter | undefined {
  return miseOrchestraRoster.find((c) => c.id === id);
}

export function getCharacterByNumber(number: number): MiseCharacter | undefined {
  return miseOrchestraRoster.find((c) => c.number === number);
}

export function getDepartmentCharacters(department: DepartmentId): MiseCharacter[] {
  return miseOrchestraRoster.filter((c) => c.department === department);
}

export function getCharactersByRarity(rarity: Rarity): MiseCharacter[] {
  return miseOrchestraRoster.filter((c) => c.rarity === rarity);
}

export function getGenesisCharacters(): MiseCharacter[] {
  return miseOrchestraRoster.filter((c) => c.genesisVariant);
}

export function getDepartmentLeads(): MiseCharacter[] {
  const leadIds = [
    "pipeline-sensei",
    "platform-sensei",
    "reliability-sensei",
    "seira-vaultkeeper",
    "roadmap-sensei"
  ];
  return miseOrchestraRoster.filter((c) => leadIds.includes(c.id));
}

export function getExecutiveLinkedCharacters(executiveName: string): MiseCharacter[] {
  return miseOrchestraRoster.filter((c) => c.executiveLink === executiveName);
}

export function getSyncNetwork(characterId: string): MiseCharacter[] {
  const source = getCharacterById(characterId);
  if (!source) return [];

  const syncSet = new Set(source.workflow.syncsWith);
  return miseOrchestraRoster.filter((c) => syncSet.has(c.name));
}

export function searchCharacters(query: string): MiseCharacter[] {
  const q = query.toLowerCase().trim();
  return miseOrchestraRoster.filter((c) => {
    return (
      c.name.toLowerCase().includes(q) ||
      c.identity.title.toLowerCase().includes(q) ||
      c.workflow.jobName.toLowerCase().includes(q) ||
      c.workflow.roleName.toLowerCase().includes(q) ||
      c.identity.nationality.toLowerCase().includes(q) ||
      c.departmentLabel.toLowerCase().includes(q)
    );
  });
}

export function getRosterStats() {
  const byDepartment = departmentOrder.map((department) => ({
    department,
    label: departmentLabels[department],
    count: getDepartmentCharacters(department).length
  }));

  const byRarity = ["common", "rare", "epic", "legendary", "mythic", "genesis"].map((rarity) => ({
    rarity,
    count: getCharactersByRarity(rarity as Rarity).length
  }));

  return {
    totalCharacters: miseOrchestraRoster.length,
    totalExecutives: miseExecutiveTier.length,
    totalGenesis: getGenesisCharacters().length,
    byDepartment,
    byRarity
  };
}
`

src/data/search.ts

`ts
import { miseOrchestraRoster } from "./roster";
import { DepartmentId, MiseCharacter, Rarity, CharacterToolType } from "./mise-orchestra-types";

export type CharacterSearchFilters = {
  department?: DepartmentId;
  rarity?: Rarity;
  nationality?: string;
  role?: string;
  tool?: CharacterToolType;
  query?: string;
};

export function filterCharacters(filters: CharacterSearchFilters): MiseCharacter[] {
  return miseOrchestraRoster.filter((character) => {
    if (filters.department && character.department !== filters.department) return false;
    if (filters.rarity && character.rarity !== filters.rarity) return false;
    if (
      filters.nationality &&
      character.identity.nationality.toLowerCase() !== filters.nationality.toLowerCase()
    ) {
      return false;
    }
    if (
      filters.role &&
      !character.identity.title.toLowerCase().includes(filters.role.toLowerCase())
    ) {
      return false;
    }
    if (filters.tool && character.visual.tool !== filters.tool) return false;

    if (filters.query) {
      const q = filters.query.toLowerCase();
      const haystack = [
        character.name,
        character.identity.title,
        character.workflow.jobName,
        character.workflow.roleName,
        character.identity.uniqueTrait,
        character.departmentLabel
      ]
        .join(" ")
        .toLowerCase();

      if (!haystack.includes(q)) return false;
    }

    return true;
  });
}
`

Character relationship graph (data side)

Rendering a true graph visually would need a graph library, but we can prep a simple graph adjacency export that you can wire later to D3/Vis.

src/data/relationship-graph.ts

`ts
import { miseOrchestraRoster } from "./roster";

export type CharacterNode = {
  id: string;
  label: string;
  department: string;
  rarity: string;
};

export type CharacterEdge = {
  from: string;
  to: string;
  reason: "syncsWith";
};

export function buildRelationshipGraph() {
  const nodes: CharacterNode[] = miseOrchestraRoster.map((c) => ({
    id: c.id,
    label: c.name,
    department: c.department,
    rarity: c.rarity
  }));

  const nodeByName = new Map<string, string>();
  miseOrchestraRoster.forEach((c) => nodeByName.set(c.name, c.id));

  const edges: CharacterEdge[] = [];

  for (const c of miseOrchestraRoster) {
    for (const targetName of c.workflow.syncsWith) {
      const targetId = nodeByName.get(targetName);
      if (!targetId) continue;
      edges.push({
        from: c.id,
        to: targetId,
        reason: "syncsWith"
      });
    }
  }

  return { nodes, edges };
}
`

1.1 src/pages/RelationshipGraphPage.tsx

`tsx
import { useMemo } from "react";
import { buildRelationshipGraph } from "../data/relationship-graph";
import { getCharacterById } from "../data/mise-orchestra-utils";
import { departmentLabels } from "../data/mise-orchestra-utils";
import { TradingCard } from "../components/TradingCard";

export function RelationshipGraphPage() {
  const { nodes, edges } = useMemo(() => buildRelationshipGraph(), []);
  const edgesBySource: Record<string, string[]> = {};

  edges.forEach((e) => {
    if (!edgesBySource[e.from]) edgesBySource[e.from] = [];
    edgesBySource[e.from].push(e.to);
  });

  return (
    <section className="page-section">
      <h1>Orchestration Relationship Graph</h1>
      <p className="muted small">
        A lightweight view of how chefs sync with one another. Each node is a chef;
        edges come from the <code>syncsWith</code> lists in their workflows.
      </p>

      <div className="graph-legend panel">
        <p className="small">
          Tip: Click a character card to open the full profile. Colors represent departments.
        </p>
      </div>

      <div className="graph-columns">
        {["pipeline", "platform", "reliability", "security", "data-insight"].map(
          (deptId) => {
            const deptNodes = nodes.filter((n) => n.department === deptId);
            return (
              <div key={deptId} className="graph-column">
                <h3 className="graph-columntitle">
                  {departmentLabels[deptId as keyof typeof departmentLabels]}
                </h3>

                <div className="graph-columnnodes">
                  {deptNodes.map((node) => {
                    const character = getCharacterById(node.id);
                    if (!character) return null;
                    const targets = edgesBySource[node.id] ?? [];

                    return (
                      <div key={node.id} className="graph-node">
                        <TradingCard character={character} />

                        {targets.length > 0 && (
                          <div className="graph-nodeedges">
                            <div className="graph-nodeedges-label">Syncs with:</div>
                            <ul>
                              {targets.map((tid) => {
                                const t = getCharacterById(tid);
                                if (!t) return null;
                                return (
                                  <li key={tid}>{t.name}</li>
                                );
                              })}
                            </ul>
                          </div>
                        )}
                      </div>
                    );
                  })}
                </div>
              </div>
            );
          }
        )}
      </div>
    </section>
  );
}
`

import { getGenesisCharacters } from "./mise-orchestra-utils";

export type MintRecord = {
  mintId: string;
  characterId: string;
  characterName: string;
  rarity: string;
  owner: string;
  timestamp: string;
};

export function getGenesisPool(): MiseCharacter[] {
  return getGenesisCharacters();
}

// naive mint id generator – swap with real hashing if needed
export function generateMintId(characterId: string): string {
  const suffix = Math.random().toString(36).slice(2, 8).toUpperCase();
  return ${characterId.toUpperCase()}-${suffix};
}

`tsx
import { useState } from "react";
import { getGenesisPool, generateMintId, MintRecord } from "../data/minting";
import { TradingCard } from "../components/TradingCard";

export function GenesisMintPage() {
  const genesisCharacters = getGenesisPool();
  const [selectedId, setSelectedId] = useState<string | undefined>(
    genesisCharacters[0]?.id
  );
  const [owner, setOwner] = useState("");
  const [mints, setMints] = useState<MintRecord[]>([]);

  const selected = genesisCharacters.find((c) => c.id === selectedId);

  function handleMint() {
    if (!selected || !owner.trim()) return;

    const mintId = generateMintId(selected.id);
    const record: MintRecord = {
      mintId,
      characterId: selected.id,
      characterName: selected.name,
      rarity: selected.rarity,
      owner: owner.trim(),
      timestamp: new Date().toISOString()
    };

    setMints((prev) => [record, ...prev]);
  }

  return (
    <section className="page-section">
      <h1>Genesis Minting Panel</h1>
      <p className="muted small">
        This panel mints collectible records for Genesis characters only.
        Connect it to a wallet/NFT backend later; for now it records simulated mints.
      </p>

      <div className="panel" style={{ marginBottom: 24 }}>
        <div className="mint-grid">
          <div className="mint-picker">
            <label>Choose a Genesis chef</label>
            <select
              value={selectedId ?? ""}
              onChange={(e) => setSelectedId(e.target.value)}
            >
              {genesisCharacters.map((c) => (
                <option key={c.id} value={c.id}>
                  #{c.number} · {c.name}
                </option>
              ))}
            </select>

            <label style={{ marginTop: 12 }}>Owner / wallet / handle</label>
            <input
              type="text"
              placeholder="@owner, wallet address, or username"
              value={owner}
              onChange={(e) => setOwner(e.target.value)}
            />

            <button
              style={{ marginTop: 12 }}
              type="button"
              disabled={!selected || !owner.trim()}
              onClick={handleMint}
            >
              Mint Genesis Card
            </button>
          </div>

          <div className="mint-preview">
            {selected && (
              <>
                <h3>Preview</h3>
                <TradingCard character={selected} />
              </>
            )}
          </div>
        </div>
      </div>

      <div className="panel">
        <h2>Recent Mints</h2>
        {mints.length === 0 && <p className="muted small">No mints yet.</p>}

        {mints.length > 0 && (
          <table className="mint-table">
            <thead>
              <tr>
                <th>Mint ID</th>
                <th>Character</th>
                <th>Rarity</th>
                <th>Owner</th>
                <th>Time (UTC)</th>
              </tr>
            </thead>

                  <td>{mint.rarity}</td>
                  <td>{mint.owner}</td>
                  <td>{new Date(mint.timestamp).toISOString().slice(0, 19)}</td>
                </tr>
              ))}
            </tbody>
          </table>
        )}
      </div>
    </section>
  );
}

1.1 Adapter: turn your graph data into React Flow nodes/edges

`ts
import { buildRelationshipGraph } from "./relationship-graph";
import { Position, Edge, Node } from "reactflow";

const deptLane: Record<string, number> = {
  pipeline: 0,
  platform: 1,
  reliability: 2,
  security: 3,
  "data-insight": 4
};

export function buildReactFlowGraph(): { nodes: Node[]; edges: Edge[] } {
  const { nodes, edges } = buildRelationshipGraph();

  const rfNodes: Node[] = nodes.map((n, index) => {
    const lane = deptLane[n.department] ?? 0;
    const x = lane  350;           // columns by department
    const y = (index % 12)  120;   // stagger vertically

    return {
      id: n.id,
      position: { x, y },
      data: {
        label: n.label,
        department: n.department,
        rarity: n.rarity
      },
      type: "default",
      sourcePosition: Position.Right,
      targetPosition: Position.Left
    };
  });

  const rfEdges: Edge[] = edges.map((e, i) => ({
    id: e-${i}-${e.from}-${e.to},
    source: e.from,
    target: e.to,
    animated: true,
    label: "syncs",
    style: { strokeWidth: 1.5 }
  }));

  return { nodes: rfNodes, edges: rfEdges };
}
`

3.1 Season‑filtered adapter

Update graph-reactflow.ts:

`ts
import { buildRelationshipGraph } from "./relationship-graph";
import { SeasonId, getSeasonCharacters } from "./lore";
import { Node, Edge, Position } from "reactflow";

const deptLane: Record<string, number> = {
  pipeline: 0,
  platform: 1,
  reliability: 2,
  security: 3,
  "data-insight": 4
};

export function buildReactFlowGraph(seasonFilter?: SeasonId): { nodes: Node[]; edges: Edge[] } {
  const { nodes, edges } = buildRelationshipGraph();

  let allowedIds: Set<string> | undefined;

  if (seasonFilter) {
    const chars = getSeasonCharacters(seasonFilter);
    allowedIds = new Set(chars.map((c) => c.id));
  }

  const filteredNodes = allowedIds
    ? nodes.filter((n) => allowedIds!.has(n.id))
    : nodes;

  const filteredEdges = edges.filter((e) => {
    if (!allowedIds) return true;
    return allowedIds.has(e.from) && allowedIds.has(e.to);
  });

  const rfNodes: Node[] = filteredNodes.map((n, index) => {
    const lane = deptLane[n.department] ?? 0;
    const x = lane  350;
    const y = (index % 12) * 120;

    return {
      id: n.id,
      position: { x, y },
      data: {
        label: n.label,
        department: n.department,
        rarity: n.rarity
      },
      type: "default",
      sourcePosition: Position.Right,
      targetPosition: Position.Left
    };
  });

  const rfEdges: Edge[] = filteredEdges.map((e, i) => ({
    id: e-${i}-${e.from}-${e.to},
    source: e.from,
    target: e.to,
    animated: true,
    label: "syncs",
    style: { strokeWidth: 1.5 }
  }));

  return { nodes: rfNodes, edges: rfEdges };
}
`

    () => (seasonFilter === "all"
      ? buildReactFlowGraph()
      : buildReactFlowGraph(seasonFilter)),
    [seasonFilter]
  );

  const [nodes, , onNodesChange] = useNodesState(initialNodes);
  const [edges, , onEdgesChange] = useEdgesState(initialEdges);

  const navigate = useNavigate();

  const onNodeClick = (: any, node: any) => {
    navigate(/character/${node.id});
  };

  return (
    <section className="page-section">
      <h1>Orchestration Graph (React Flow)</h1>

      <div className="panel" style={{ marginBottom: 10 }}>
        <div className="field" style={{ maxWidth: 260 }}>
          <label>Season overlay</label>
          <select
            value={seasonFilter}
            onChange={(e) =>
              setSeasonFilter(
                e.target.value === "all" ? "all" : (e.target.value as SeasonId)
              )
            }
          >
            <option value="all">All seasons</option>
            {seasons.map((s) => (
              <option key={s.id} value={s.id}>
                {s.title}
              </option>
            ))}
          </select>
        </div>
      </div>

      <div className="panel" style={{ height: "70vh", padding: 0, overflow: "hidden" }}>
        <ReactFlow
          nodes={nodes}
          edges={edges}
          onNodesChange={onNodesChange}
          onEdgesChange={onEdgesChange}
          fitView
          onNodeClick={onNodeClick}
          nodesDraggable
        >
          <MiniMap
            nodeStrokeColor={(node) => deptColor[node.data.department] ?? "#6B7280"}
            nodeColor={(node) => deptColor[node.data.department] ?? "#E5E7EB"}
          />
          <Controls />
          <Background gap={16} color="#E5E7EB" />
        </ReactFlow>
      </div>
    </section>
  );
}
`
