/**
 * Middleware: Validate session token
 */
function requireSession(req, res, next) {
  const token = req.headers.authorization?.replace('Bearer ', '');
  if (!token) {
    return res.status(401).json({ error: 'Missing session token' });
  }
  
  const session = sessions.get(token);
  if (!session || Date.now() > session.expiresAt) {
    sessions.delete(token);
    return res.status(401).json({ error: 'Invalid or expired session' });
  }
  
  req.session = session;
  next();
}

/**
 * Wallet Authentication Module
 * Nonce-based sign-in with Ethereum message verification
 */

import { verifyMessage, getAddress } from 'ethers';
import crypto from 'crypto';

// In-memory nonce store (use Redis in production)
const nonceStore = new Map();
const NONCE_EXPIRY = 5 * 60 * 1000; // 5 minutes

/**
 * Create a nonce for wallet login
 * @param {string} address - Ethereum address
 * @returns {string} UUID nonce
 */
export function createNonce(address) {
  const normalizedAddress = getAddress(address); // checksum address
  const nonce = crypto.randomUUID();
  
  // Clean old entries
  const now = Date.now();
  for (const [key, entry] of nonceStore.entries()) {
    if (entry.expiresAt < now) nonceStore.delete(key);
  }
  
  nonceStore.set(normalizedAddress.toLowerCase(), {
    nonce,
    createdAt: now,
    expiresAt: now + NONCE_EXPIRY
  });
  
  return nonce;
}

/**
 * Verify wallet login signature
 * @param {string} address - Ethereum address
 * @param {string} message - Signed message
 * @param {string} signature - Signature
 * @returns {boolean} Validity
 */
export function verifyWalletLogin(address, message, signature) {
  try {
    const normalizedAddress = address.toLowerCase();
    const stored = nonceStore.get(normalizedAddress);
    
    if (!stored) {
      console.warn('No nonce found for address:', address);
      return false;
    }
    
    if (Date.now() > stored.expiresAt) {
      console.warn('Nonce expired for address:', address);
      nonceStore.delete(normalizedAddress);
      return false;
    }
    
    // Verify signature recovers the expected address
    const recoveredAddress = verifyMessage(message, signature);
    
    if (recoveredAddress.toLowerCase() !== normalizedAddress) {
      console.warn('Signature address mismatch');
      return false;
    }
    
    // Verify nonce is in message
    if (!message.includes(stored.nonce)) {
      console.warn('Nonce not found in message');
      return false;
    }
    
    // Verify message format (prevent replay attacks)
    const expectedPrefix = 'Sign in to Clickable Character';
    if (!message.startsWith(expectedPrefix)) {
      console.warn('Invalid message format');
      return false;
    }
    
    // Clean up used nonce
    nonceStore.delete(normalizedAddress);
    
    return true;
  } catch (err) {
    console.error('Wallet verification error:', err);
    return false;
  }
}

/**
 * Generate sign-in message
 * @param {string} nonce - Random nonce
 * @param {string} domain - App domain
 * @returns {string} Formatted message
 */
export function generateSignInMessage(nonce, domain = 'clickable-character.app') {
  return `Sign in to Clickable Character

Domain: ${domain}
Nonce: ${nonce}
Timestamp: ${new Date().toISOString()}

This signature proves you own this wallet address. It does not cost any gas.`;
}

/**
 * NFT Ownership Verification Module
 * Checks on-chain token ownership for access control
 */

import { Contract, JsonRpcProvider } from 'ethers';

// Standard ERC721/ERC1155 ABI fragments
const ERC721_ABI = [
  'function balanceOf(address owner) view returns (uint256)',
  'function ownerOf(uint256 tokenId) view returns (address)',
  'function totalSupply() view returns (uint256)'
];

const ERC1155_ABI = [
  'function balanceOf(address account, uint256 id) view returns (uint256)',
  'function balanceOfBatch(address[] accounts, uint256[] ids) view returns (uint256[])'
];

/**
 * Check if address owns NFT from contract
 * @param {Object} params
 * @param {string} params.rpcUrl - JSON-RPC endpoint
 * @param {string} params.contractAddress - NFT contract address
 * @param {string[]} params.abi - Contract ABI (optional, uses standard)
 * @param {string} params.ownerAddress - Wallet address to check
 * @param {string} params.tokenStandard - 'ERC721' or 'ERC1155'
 * @param {string} params.tokenId - Specific token ID (for ERC1155)
 * @returns {Promise<boolean>}
 */
export async function hasRequiredNft({
  rpcUrl,
  contractAddress,
  abi,
  ownerAddress,
  tokenStandard = 'ERC721',
  tokenId = null
}) {
  try {
    const provider = new JsonRpcProvider(rpcUrl);
    const contractAbi = abi || (tokenStandard === 'ERC1155' ? ERC1155_ABI : ERC721_ABI);
    const contract = new Contract(contractAddress, contractAbi, provider);
    
    let balance;
    
    if (tokenStandard === 'ERC1155' && tokenId !== null) {
      // Check specific ERC1155 token balance
      balance = await contract.balanceOf(ownerAddress, tokenId);
    } else {
      // Check ERC721 or general ERC1155 balance
      balance = await contract.balanceOf(ownerAddress);
    }
    
    return balance > 0n;
  } catch (err) {
    console.error('NFT check failed:', err.message);
    // Fail closed - if we can't verify, deny access
    return false;
  }
}

/**
 * Get NFT metadata (if available)
 * @param {Object} params
 * @param {string} params.rpcUrl
 * @param {string} params.contractAddress
 * @param {string} params.tokenId
 * @returns {Promise<Object|null>}
 */
export async function getNftMetadata({ rpcUrl, contractAddress, tokenId }) {
  try {
    const provider = new JsonRpcProvider(rpcUrl);
    const contract = new Contract(contractAddress, [
      'function tokenURI(uint256 tokenId) view returns (string)'
    ], provider);
    
    const tokenURI = await contract.tokenURI(tokenId);
    
    // Handle base64-encoded JSON metadata
    if (tokenURI.startsWith('data:application/json;base64,')) {
      const base64 = tokenURI.replace('data:application/json;base64,', '');
      const json = Buffer.from(base64, 'base64').toString('utf8');
      return JSON.parse(json);
    }
    
    // Handle IPFS
    if (tokenURI.startsWith('ipfs://')) {
      const gateway = 'https://ipfs.io/ipfs/';
      const httpUrl = tokenURI.replace('ipfs://', gateway);
      const response = await fetch(httpUrl);
      return await response.json();
    }
    
    // Handle HTTP(S)
    if (tokenURI.startsWith('http')) {
      const response = await fetch(tokenURI);
      return await response.json();
    }
    
    return { tokenURI };
  } catch (err) {
    console.error('Metadata fetch failed:', err);
    return null;
  }
}

/**
 * Check multiple NFT conditions
 * @param {Object[]} requirements - Array of NFT requirements
 * @param {string} ownerAddress
 * @returns {Promise<Object>} Results per requirement
 */
export async function checkNftRequirements(requirements, ownerAddress) {
  const results = {};
  
  for (const req of requirements) {
    const key = req.policyId || req.contractAddress;
    results[key] = await hasRequiredNft({
      ...req,
      ownerAddress
    });
  }
  
  return results;
}

/**
 * Workflow Execution Engine
 * Resolves and executes allowlisted workflow modules
 */

import { createRequire } from 'module';
import path from 'path';
import { fileURLToPath } from 'url';

const __dirname = path.dirname(fileURLToPath(import.meta.url));

// Workflow registry - maps action IDs to module paths
// This is loaded from workflow/registry.json
let registry = {};

/**
 * Load workflow registry
 */
export async function loadRegistry() {
  try {
    const registryPath = path.join(__dirname, '../workflow/registry.json');
    const module = await import(registryPath, { assert: { type: 'json' } });
    registry = module.default || module;
    console.log('Workflow registry loaded:', Object.keys(registry));
  } catch (err) {
    console.error('Failed to load workflow registry:', err);
    registry = {};
  }
}

/**
 * Execute allowlisted workflow
 * @param {string} actionId - Registered action identifier
 * @param {Object} context - Execution context (address, params, etc.)
 * @returns {Promise<any>} Workflow result
 */
export async function executeWorkflow(actionId, context) {
  // Security: Verify action is in registry
  if (!Object.hasOwn(registry, actionId)) {
    throw new Error(`Action '${actionId}' not found in workflow registry`);
  }
  
  const modulePath = registry[actionId];
  
  try {
    // Resolve module path (relative to workflow directory)
    const fullPath = path.join(__dirname, '../workflow', modulePath);
    
    // Dynamic import with assertion for security
    const workflowModule = await import(fullPath);
    
    // Get the handler function
    const handler = workflowModule[actionId] || workflowModule.default;
    
    if (typeof handler !== 'function') {
      throw new Error(`Workflow '${actionId}' has no exported handler`);
    }
    
    // Execute with timeout protection
    const result = await Promise.race([
      handler(context),
      new Promise((_, reject) => 
        setTimeout(() => reject(new Error('Workflow timeout')), 30000)
      )
    ]);
    
    return {
      success: true,
      actionId,
      result,
      executedAt: Date.now()
    };
    
  } catch (err) {
    console.error(`Workflow execution failed for ${actionId}:`, err);
    throw err;
  }
}

/**
 * Get available workflows for a user based on their NFT access
 * @param {Object} nftAccess - User's NFT holdings
 * @returns {Object} Filtered workflow registry
 */
export function getAvailableWorkflows(nftAccess = {}) {
  // In production, filter based on nftAccess tiers
  return Object.entries(registry).reduce((acc, [key, path]) => {
    acc[key] = { path, requiresNft: key.includes('premium') || key.includes('pro') };
    return acc;
  }, {});
}

/**

 */
function drawHotspotOverlays() {
  if (!verifiedManifest) return;
  
  hotspots.forEach(h => {
    ctx.save();
    ctx.strokeStyle = h.requiresNft ? '#ff6b6b' : '#4ecdc4';
    ctx.lineWidth = 2;
    ctx.setLineDash([5, 5]);
    
    if (h.shape.type === 'circle') {
      ctx.beginPath();
      ctx.arc(h.shape.x, h.shape.y, h.shape.r, 0, Math.PI * 2);
      ctx.stroke();
    } else if (h.shape.type === 'rect') {
      ctx.strokeRect(h.shape.x, h.shape.y, h.shape.w, h.shape.h);
    }
    
    // Label
    ctx.fillStyle = h.requiresNft ? '#ff6b6b' : '#4ecdc4';
    ctx.font = '12px sans-serif';
    ctx.fillText(h.name, h.shape.x, h.shape.y - 5);
    
    ctx.restore();
  });
}

/**
 * Hit test for hotspot detection
 */
function hitTest(x, y, hotspot) {
  const h = hotspot.shape;
  
  switch (h.type) {
    case 'circle': {
      const dx = x - h.x;
      const dy = y - h.y;
      return (dx * dx + dy * dy) <= (h.r * h.r);
    }
    case 'rect': {
      return x >= h.x && x <= h.x + h.w && 
             y >= h.y && y <= h.y + h.h;
    }
    case 'poly': {
      // Simple polygon hit test using ray casting
      const points = h.points;
      let inside = false;
      for (let i = 0, j = points.length - 1; i < points.length; j = i++) {
        const xi = points[i][0], yi = points[i][1];
        const xj = points[j][0], yj = points[j][1];
        const intersect = ((yi > y) !== (yj > y)) &&
          (x < (xj - xi) * (y - yi) / (yj - yi) + xi);
        if (intersect) inside = !inside;
      }
      return inside;
    }
    default:
      return false;
  }
}

/**
 * Handle canvas click
 */
async function handleCanvasClick(e) {
  if (!verifiedManifest) {
    alert('Manifest not verified. Cannot interact.');
    return;
  }
  
  const rect = canvas.getBoundingClientRect();
  const scaleX = canvas.width / rect.width;
  const scaleY = canvas.height / rect.height;
  const x = (e.clientX - rect.left) * scaleX;
  const y = (e.clientY - rect.top) * scaleY;
  
  // Find clicked hotspot
  const hotspot = hotspots.find(h => hitTest(x, y, h));
  if (!hotspot) return;
  
  // Check NFT requirement
  if (hotspot.requiresNft && (!session?.nftAccess?.hasBasic)) {
    alert('This feature requires NFT ownership. Please connect a wallet with the required token.');
    return;
  }
  
  // Get user consent
  const consent = confirm(
    `Run action: "${hotspot.name}" (${hotspot.actionId})?\n\n` +
    `This will execute an approved workflow.`
  );
  
  if (!consent) return;
  
  // Execute workflow
  await executeHotspotAction(hotspot);
}

/**
 * Execute hotspot action via API
 */
async function executeHotspotAction(hotspot) {
  try {
    updateStatus('Authorizing...', 'loading');
    
    // Step 1: Authorize action
    const authRes = await fetch(`${API_URL}/api/authorize-action`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${session?.token || 'anonymous'}`
      },
      body: JSON.stringify({
        hotspotId: hotspot.id,
        actionId: hotspot.actionId,
        manifestId: manifest.asset?.sha256?.slice(0, 16)
      })
    });
    
    const auth = await authRes.json();
    
    if (!auth.allowed) {
      throw new Error(auth.reason || 'Action not authorized');
    }
    
    updateStatus('Executing workflow...', 'loading');
    
    // Step 2: Execute workflow
    const execRes = await fetch(`${API_URL}/api/execute-workflow`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${session?.token || 'anonymous'}`
      },
      body: JSON.stringify({
        actionId: hotspot.actionId,
        context: auth.context,
        params: { hotspotId: hotspot.id, name: hotspot.name }
      })
    });
    
    const result = await execRes.json();
    
    if (result.ok) {
      updateStatus(`Success: ${hotspot.actionId}`, 'success');
      showResult(result);
    } else {
      throw new Error(result.error);
    }
    
  } catch (err) {
    console.error('Action failed:', err);
    updateStatus(`Failed: ${err.message}`, 'error');
  }
}

 */
function handleMouseMove(e) {
  const rect = canvas.getBoundingClientRect();
  const scaleX = canvas.width / rect.width;
  const scaleY = canvas.height / rect.height;
  const x = (e.clientX - rect.left) * scaleX;
  const y = (e.clientY - rect.top) * scaleY;
  
  const hovering = hotspots.some(h => hitTest(x, y, h));
  canvas.style.cursor = hovering ? 'pointer' : 'default';
}

/**
 * Connect wallet and authenticate
 */
async function connectWallet() {
  if (!window.ethereum) {
    alert('Please install MetaMask or another Web3 wallet');
    return;
  }
  
  try {
    updateStatus('Connecting wallet...', 'loading');
    
    // Request accounts
    const [address] = await window.ethereum.request({ 
      method: 'eth_requestAccounts' 
    });
    
    // Get nonce from server
    const nonceRes = await fetch(`${API_URL}/api/auth/nonce`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ address })
    });
    
    const { nonce } = await nonceRes.json();
    
    // Sign message
    const message = `Sign in to Clickable Character\nNonce: ${nonce}`;
    const signature = await window.ethereum.request({
      method: 'personal_sign',
      params: [message, address]
    });
    
    // Verify with server
    const verifyRes = await fetch(`${API_URL}/api/auth/verify`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ address, message, signature })
    });
    
    const sessionData = await verifyRes.json();
    
    // Store session
    session = {
      token: sessionData.token,
      address: sessionData.address,
      nftAccess: sessionData.nftAccess
    };
    
    localStorage.setItem('cc_session', JSON.stringify(session));
    
    updateStatus(`Connected: ${address.slice(0, 6)}...${address.slice(-4)}`, 'success');
    
    // Refresh UI
    drawHotspotOverlays();
    
  } catch (err) {
    console.error('Wallet connection failed:', err);
    updateStatus('Connection failed: ' + err.message, 'error');
  }
}
